\section{Completeness and approximation}

\begin{result}
\subsection{Cook--Levin and its complement}
SAT is NP-complete and UNSAT is coNP-complete under tidy polynomial-time
many-one reductions, using the library's encoding \lean{encCNF} of CNF formulas.
Reduction efficiency is proved, not assumed. SAT is also NP-complete
when every clause has exactly three literal occurrences:
repeated literals are permitted. This does not establish the variant requiring
three distinct literals per clause.\citep[Section 2.3]{AroraBarak09}

\begin{result}
\subsection{Parsimonious Cook--Levin and complete counting problems}
The Cook--Levin tableau is parsimonious: for a deterministic multi-tape machine $M$, input
prefix $u$, certificate length $L$ and step bound $T$, the tableau has exactly as many
satisfying assignments (of the variables occurring in it) as there are words
$w\in\{0,1\}^L$ such that $M$ on $uw$ has halted with output $1$ after $T$ steps. Hence the
existing Cook--Levin reduction of a verifier $V$ maps an instance $a$ to a CNF formula with
exactly $|\{w\in\{0,1\}^{p(|a|)} : V(a,w)=1\}|$ satisfying assignments, and $\#\mathrm{SAT}$
is $\#\cc{P}$-complete under parsimonious polynomial-time reductions $f(a)=\#\mathrm{SAT}(r(a))$,
for the verifier-based $\#\cc{P}$ (certificates of exact polynomial length) and every
injective instance encoding. The count ranges over assignments of the occurring variables.
Consequently, under polynomial-time many-one reductions: formulas with an odd number of
satisfying assignments form a $\oplus\cc{P}$-complete problem; unique satisfiability is
$\cc{US}$-complete; a count not divisible by $k$ is $\cc{Mod}_k\cc{P}$-complete and its
complement $\cc{coMod}_k\cc{P}$-complete, for every modulus $k$ (including the degenerate
moduli $0$ and $1$ of the total definition); pairs $(\varphi,m)$, $m$ in binary, with
$\#\varphi\ge m$ form a $\cc{PP}$-complete problem, and those with $\#\varphi=m$ a
$\cc{C_{=}P}$-complete problem whose complement is $\cc{coC_{=}P}$-complete. All classes are the
library's verifier-based definitions; MAJSAT itself is not treated.

\begin{result}
\subsection{DP and SAT--UNSAT completeness}
The difference class $\cc{DP}$ consists of languages $B\cap C$ with
$B\in\cc{NP}$ and $C\in\cc{coNP}$, equivalently differences of two NP
languages. For any NP-complete $B$ and coNP-complete $C$, the product language
$B\times C$ is DP-complete under polynomial-time many-one reductions, using
the product encoding. In particular, pairs of CNF formulas $(\varphi,\psi)$
with $\varphi$ satisfiable and $\psi$ unsatisfiable form a DP-complete problem.
The theorem quantifies over all injectively encoded source types; the two
component reductions and their pairing have certified polynomial time.
The definition permits distinct NP and coNP components; it does not require
the resulting language to belong to both classes.\citep{Papadimitriou94}
\declat{4f6bdaf35ebe60aec6d1063ae513ac342df13589}{Complexity/Computability/Complexity/Classes/DP.lean}{87}{Cslib.Complexity.NPComplete.prod_coNPComplete}
\declat{4f6bdaf35ebe60aec6d1063ae513ac342df13589}{Complexity/Computability/Complexity/Problems/SATUNSAT.lean}{40}{Cslib.Complexity.SATUNSAT_dpComplete}
The complement class coDP consists of unions of an NP language and a coNP
language; it is closed under union and polynomial-time many-one reductions.

\begin{result}
\subsection{The Boolean hierarchy over NP}
Using the difference-hierarchy convention,
\[
\cc{BH}_0=\{\varnothing\},\qquad
\cc{BH}_{n+1}=\{B\setminus D:B\in\cc{NP},\ D\in\cc{BH}_n\}.
\]
We prove $\cc{BH}_1=\cc{NP}$ and $\cc{BH}_2=\cc{DP}$, monotonicity of
levels, complement membership one level higher, and polynomial-time many-one
reduction closure at each fixed level. Consequently the union of the finite
levels is closed under complement. The construction and inverse-image theorem
are generic over the base class of sets; NP supplies the complexity instance.
This follows the difference presentation of the Boolean hierarchy; no separation
between levels or hierarchy-collapse theorem is asserted.\citep{CaiEtAl1988}

\begin{result}
\subsection{Graph and number problems are NP-complete}
Three-colorability, undirected Hamiltonian cycle, binary subset sum, and the
budget decision version of traveling salesman are NP-complete under certified
polynomial-time many-one reductions. Subset sum selects entries of a list of
natural numbers, allowing repeated values; its binary encoding does not give
strong NP-hardness. TSP uses binary natural distances and budget, and does not
require symmetry or the triangle inequality.\citep{Karp1972,GareyJohnson1979,Sipser2013}

\begin{result}
\subsection{Exponential lower bounds for graph problems under ETH}
Assume the exponential time hypothesis. Then for each of the following problems there is
$\delta>0$ such that it is not decidable within time $2^{\delta\mu}$ times a polynomial in the
length of the encoded instance: three-colorability, independent set and vertex cover (graph with
binary target), and directed and undirected Hamiltonian cycle and path (graph with endpoints for
paths), all with $\mu=|V|+|E|$; independent set and clique with $\mu=|V|$. A graph is a list of
declared vertices and a list of edges, and $|V|$, $|E|$ are the list lengths with repetitions;
they bound the distinct counts, so the bounds for distinct counts follow. The predicate negated
allows the decider to depend on $\delta$. Each bound composes a polynomial-time reduction from
3-CNF formulas whose output measure is linear in the number $m$ of clauses with the
sparsification lemma, under which ETH excludes $2^{o(m)}$ for 3SAT. Three-colorability uses the
gadget reduction (at most $(24k+12)m+9$ vertices and edges for clauses of width at most $k$);
independent set, vertex cover and clique use the variable-edge reduction of Garey and Johnson
(at most $(k^2+5k)m$), not Karp's occurrence graph, which can have quadratically many edges.
Hamiltonian problems use a variant of Sipser's diamond construction with one diamond per
distinct variable and one slot per occurrence; its directed path graph has at most $(23k+1)m+1$
vertices and arcs in total, and the cycle and undirected graphs obtained by closing and splitting
it stay linear (at most $(115k+5)m+10$ for undirected Hamiltonian cycle). The correctness of
diamond graphs is proved once for arbitrary row layouts, and Sipser's construction, with
$\Theta(m^2)$ vertices in the worst case, is an instance of it. Garey and Johnson's reduction from
vertex cover to Hamiltonian cycle is not used: its selector edges are $\Theta(m^2)$ on these
instances.

\begin{result}
\textbf{Circuit evaluation.}
Given a one-output De Morgan circuit and an input of the matching arity, deciding
whether its output is true is in P. Instances are well formed by their type;
\lean{encCircuitValue} encodes the circuit and input bits. The standard
P-completeness theorem under logspace reductions is not yet formalized here.
\citep{Ladner1975}
\decl{Complexity/Computability/Complexity/Problems/CircuitValue.lean}{105}{Cslib.Complexity.CircuitValue.mem_P}
\end{result}

\begin{result}
\subsection{The PCP theorem}
\[
\cc{NP}=\cc{PCP}(O(\log n),O(1)).
\]
The verifier is polynomial-time and nonadaptive, has perfect completeness and
soundness at most $1/2$, and uses logarithmically many random bits and constantly
many queries. In the Lean equality the bound functions are
$\lfloor\log_2 n\rfloor+1$ (with Lean's natural-number logarithm) and $1$;
the class definition allows constant multiples of both. It does not assert
one query.\citep{Dinur2007}
\decl{Complexity/Computability/PCP/Completeness.lean}{81}{Cslib.Complexity.NP_eq_PCP_log}
\end{result}

\begin{result}
\subsection{A constant gap for MAX-3SAT}
It is NP-hard, under polynomial-time promise reductions, to distinguish a
satisfiable formula with clauses of at most three literals from one in which
every assignment satisfies at most the fraction
\[
1-\frac{1}{3\cdot 2^{77}}
\]
of its clauses. Clauses are counted with multiplicity. This is the explicit gap
obtained from the formalized PCP construction; no optimal approximation threshold
is asserted.\citep{AroraLundMotwaniSudanSzegedy1998,Dinur2007}
\decl{Complexity/Computability/PCP/Approximation/MaxThreeSAT.lean}{469}{Cslib.Complexity.GapThreeSAT_npHard}
\end{result}

\begin{result}
\subsection{PCP without random bits}
For the same nonadaptive polynomial-time verifier model,
\[
\cc{PCP}(0,0)=\cc P,\qquad
A\in\cc{NP}\ \Longleftrightarrow\
\exists k\in\mathbb N,\ A\in\cc{PCP}\bigl(0,n\mapsto(n+1)^k\bigr).
\]
The class definition allows constant multiples of the query bound. With no
random bits, perfect completeness and soundness at most $1/2$ force deterministic
acceptance of an appropriate proof for a member and rejection of every proof
for a nonmember. These identities hold for each injective bit-string encoding.
\citep[Chapter 11]{AroraBarak09}
\decl{Complexity/Computability/Complexity/Classes/PCP/ZeroRandomness.lean}{58}{Cslib.Complexity.PCP_zero_zero}
\decl{Complexity/Computability/Complexity/Classes/PCP/ZeroRandomness.lean}{146}{Cslib.Complexity.mem_NP_iff_exists_PCP_zero}
\end{result}

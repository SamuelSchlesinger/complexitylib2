\section{Proof complexity}

\begin{result}
\subsection{Cook--Reckhow characterization}
NP equals coNP for every encoded instance type if and only if UNSAT has a
polynomially bounded Cook--Reckhow proof system: a polynomial-time function from
bit strings onto unsatisfiable CNF formulas, with a preimage of polynomial length
for every such formula. Polynomial boundedness asserts that short proofs exist;
it does not give a polynomial-time procedure for finding them.
\citep{CookReckhow1979}
\decl{Complexity/Logics/Propositional/ProofSystem.lean}{320}{Cslib.Complexity.ProofSystem.NP_eq_coNP_iff_exists_polynomiallyBounded}
\end{result}

\begin{result}
\subsection{Resolution size--width tradeoff}
If a CNF formula has initial clause width at most $k$, at most $N$ variables,
and a resolution refutation with at most $s$ lines, it has a refutation of some
integer width $w$ satisfying
\[
w\le k+2\sqrt{2N\ln s}+2.
\]
Thus a large minimum refutation width forces a large refutation size. The
statement concerns proof certificates, not the runtime of a resolution or DPLL
solver.\citep{BenSassonWigderson2001}
\decl{Complexity/Logics/Propositional/Resolution/SizeWidth.lean}{331}{Cslib.Logic.PL.CNF.exists_refutableInWidth_le_of_refutableInSize}
\end{result}

\begin{result}
\subsection{Tseitin formulas are hard for resolution}
Let $G$ be a finite multigraph on $V$ with edge set $E$ (self-loops and parallel
edges allowed) and $\chi:V\to\{0,1\}$ a charge. The Tseitin formula $T(G,\chi)$
has a variable per edge and, for each vertex $v$, at most $2^{d(v)}$ clauses of
width $d(v)$ stating that the parity of the non-loop edges at $v$ is $\chi(v)$
($d(v)$ = number of non-loop edges at $v$). It is unsatisfiable whenever
$\sum_v\chi(v)$ is odd; on a connected $G$ it is satisfiable exactly when
$\sum_v\chi(v)$ is even. Let $e(G)$ be the least number of edges between a set
$S$ with $|V|/3\le|S|\le2|V|/3$ and its complement ($e(G)=0$ if $|V|\le1$). On a
connected $G$ every resolution refutation of $T(G,\chi)$ has width (distinct literals in
a line) at least $e(G)$; if moreover every vertex has at most $k$ non-loop edges,
$|E|\le N$ and $k+2\le e(G)$, every refutation has at least $\exp((e(G)-k-2)^2/(8N))$
lines. For the multigraph of a rotation map, $e(G)\ge h(G)|V|/3$ with $h$ the edge
expansion not normalized by the degree. Consequently, for every $n\ge 4392000$ there is an
unsatisfiable CNF with at most $120n$ variables and $2^{120}\cdot n$ clauses of width at most
$120$ all of whose resolution refutations have at least $\exp(n/(960\cdot36000^2))$ lines.
The formula is the Tseitin formula, with one charged vertex, of a $120$-regular expander
whose existence is proved non-constructively; the constants are far from optimal, and the
statement concerns proof size, not solver runtime.
\citep{Tseitin1983,Urquhart1987,BenSassonWigderson2001,Jukna2012}

\begin{result}
\subsection{Haken's pigeonhole lower bound for resolution}
For $m$ pigeons and $n$ holes, the propositional pigeonhole formula $\mathrm{PHP}^m_n$
on variables $\mathrm{Fin}\,m\times\mathrm{Fin}\,n$ has $m$ pigeon clauses
$\bigvee_{j<n} x_{i,j}$ of length $n$ and, for each hole $j<n$ and distinct pigeons
$i\ne i'$, a hole clause $\neg x_{i,j}\vee\neg x_{i',j}$ of length $2$ (at most
$m+nm^2$ clauses on at most $mn$ variables). It is satisfiable if and only if
$m\le n$; in particular, $\mathrm{PHP}^{n+1}_n$ is unsatisfiable for every $n$.
Every resolution refutation of $\mathrm{PHP}^{n+1}_n$ contains a clause $c$ whose
set of critical-assignment satisfaction witnesses $(i,j)$ satisfies
$2n^2\le 9\,|\mathrm{satPairs}(c)|$. Eliminating such clauses by $\lfloor n/3\rfloor$
greedy matching restrictions shows that if $\mathrm{PHP}^{n+1}_n$ has a resolution
refutation with at most $s$ lines then $12^{\lfloor n/3\rfloor}\le 11^{\lfloor n/3\rfloor}s$
for all $n\ge 8$, and $2^{\lfloor n/24\rfloor}\le s$ for all $n\in\mathbb{N}$ (and
hence $2^{\lfloor n/24\rfloor}\le |d|$ for every tree-like resolution refutation $d$).
The formalization uses the Beame--Pitassi simplification of Haken's bottleneck
counting argument rather than the size--width tradeoff, whose hypothesis on initial
clause width is uninformative for the width-$n$ pigeon clauses of $\mathrm{PHP}^{n+1}_n$;
it covers the non-onto formulation without hole-surjectivity clauses, and bounds
proof size rather than solver runtime.
\citep{Haken1985,BeamePitassi1996,Jukna2012}

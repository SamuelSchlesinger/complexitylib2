\section{Communication and formulas}

\begin{result}
\subsection{Karchmer--Wigderson correspondence}
For a Boolean function on $n>0$ bits, minimum De Morgan formula depth, counting
AND/OR gates and making negations free, equals deterministic communication
complexity of the following relation: Alice receives a true input, Bob a false
input, and they must output a coordinate on which the inputs differ. The equality
uses the library's worst-case protocol depth and extended-natural complexity.
\citep{KarchmerWigderson1990}
\decl{Complexity/Computability/Circuit/Boolean/KarchmerWigderson.lean}{282}{Cslib.Circuits.Boolean.formulaDepth_eq_ecomplexity}
\end{result}

\begin{result}
\subsection{Yao's minimax principle}
For finite input spaces and a nonempty output space, the public-coin communication
complexity of a relation with worst-case error at most $\varepsilon$ equals the
supremum, over input distributions, of the minimum deterministic communication
cost with distributional error at most $\varepsilon$. Both directions are proved,
including the minimax step; the result is not only the easy averaging lower bound.
Costs are worst-case numbers of communicated bits.\citep{Yao1977,KushilevitzNisan1997}
\decl{Complexity/Computability/Communication/Distributional.lean}{158}{Cslib.Communication.PublicCoin.ecomplexity_eq_iSup_distributionalComplexity}
\end{result}

\begin{result}
\subsection{Newman's theorem}
If each party's input space has cardinality at most $2^n$, then for every
$\delta>0$, private-coin communication complexity at error $\varepsilon+\delta$
is at most public-coin complexity at error $\varepsilon$ plus
\[
\left\lceil\log_2\left\lceil n/\delta^2\right\rceil\right\rceil
\]
bits. The formal expression uses natural ceiling and \lean{Nat.clog}, including
its zero-argument convention. The shared finite sample of protocols is chosen
existentially, without a certified procedure to construct it.\citep{Newman1991}
\decl{Complexity/Computability/Communication/Newman.lean}{168}{Cslib.Communication.PrivateCoin.ecomplexity_le_add_clog}
\end{result}

\begin{result}
\subsection{Randomized communication complexity of disjointness}
Deciding whether two $n$-bit strings have disjoint supports needs $\Omega(n)$ bits with
bounded error: for every $n$, the public-coin communication complexity $R$ of
disjointness with worst-case error $1/3$ satisfies $n\le 2^{16}R+2^{12}$, and the same
holds for private coins; with error $\varepsilon\le 1/64$, $n\le 2^{9}R_\varepsilon+2^{12}$.
Public-coin protocols are arbitrary distributions over deterministic protocols, with cost
the largest depth in the support. The proof is Razborov's corruption argument with a
variant of his distribution, using all subsets of the parts rather than fixed-size ones: a
uniformly random partition $(i,T_x,T_y)$ of $2m+1$ points into a point and two
$m$-sets, uniformly random subsets $x\subseteq T_x$, $y\subseteq T_y$, and with
probability $1/2$ the point $i$ added to both sets. On every rectangle the intersecting
pairs carry at least $1/18$ of the mass of the disjoint pairs, up to $(27/32)^r/2$ when
$48r\le m+1$. Error $1/3$ is reduced to $1/64$ by majority votes of $72$ independent runs.
Costs are worst-case numbers of communicated bits, and the constants are not optimized.
\citep{KalyanasundaramSchnitger1992,Razborov1992,KushilevitzNisan1997}

\begin{result}
\subsection{Monotone depth of bipartite perfect matching}
Every monotone formula (AND/OR gates of fan-in two, constants free) deciding whether a
subgraph of $K_{n,n}$, given by its $n^2$ edge indicators, has a perfect matching has depth at
least $(n-2^{14})/2^{13}$, and the monotone Karchmer--Wigderson game of bipartite perfect
matching needs that many bits. The proof is a randomized reduction from the disjointness bound
of the preceding entry at error $1/64$: Alice's set becomes a perfect matching on three-vertex
gadgets, Bob's set a Hall-violating graph, and after a uniformly random relabelling of both
sides the edge found reveals an intersection with probability at least $1/2$, by a symmetry of
both graphs; for $0<m$, $2m\le n$ and $k>0$, public-coin disjointness of $m$ bits at error
$2^{-k}$ costs at most $k(c+1)$ bits given a game protocol of $c$ bits. Raz and Wigderson proved
the bound for perfect matching in general graphs; this entry covers the bipartite case only,
and the constants are not optimized.\citep{RazWigderson1992,KushilevitzNisan1997}

\begin{result}
\subsection{Finite-set equality and inner-product rank}
Equality and inequality on a finite set $X$ with $|X|\ge2$ each have
deterministic communication complexity exactly $\lceil\log_2|X|\rceil+1$.
The protocol model labels leaves by the answer, so the transcript determines
the output; the bound includes the bit announcing the comparison result.

For inner product modulo two on two $n$-bit inputs, the sign matrix has rank
$2^n$ and the $0/1$ communication matrix has rank $2^n-1$ over any field of
characteristic different from two. A deterministic protocol computing it has
at least $2^n-1$ leaves labelled true. These are exact-protocol and algebraic
bounds, not randomized communication lower bounds.\citep{KushilevitzNisan1997}

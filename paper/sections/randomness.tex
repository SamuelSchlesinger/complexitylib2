\begin{result}
\subsection{Sipser--G\'acs--Lautemann}
\[
\cc{BPP}\subseteq\Sigma^p_2\cap\Pi^p_2.
\]
For the library's bounded-error polynomial-time class, membership has both an
existential--universal and a universal--existential characterization with
polynomially bounded bit-string witnesses and a polynomial-time predicate.
\citep{Sipser1983,Lautemann1983}

\begin{result}
As a consequence of the hierarchy collapse, if $\cc{NP}\subseteq\cc{P}$ for
every type and injective bit-string encoding, then $\cc{BPP}=\cc{P}$ for every
such encoded problem class. The formal premise quantifies over all encodings,
rather than assuming equality only at one chosen encoding.\citep{AroraBarak09}
\decl{Complexity/Computability/Complexity/Classes/BPP/Collapse.lean}{45}{Cslib.Complexity.BPP_eq_P_of_NP_subset_P}
\end{result}

\begin{result}
\subsection{Arthur--Merlin classes}
For every fixed injective encoding,
\[
\cc{NP}\cup\cc{BPP}\subseteq\cc{MA}\subseteq\cc{AM}\subseteq\Pi^p_2,\qquad
\cc{MA}\subseteq\Sigma^p_2\cap\Pi^p_2,\qquad \cc{MA}\subseteq\cc{SBP}.
\]
MA and AM are two-message public-coin classes given by deterministic polynomial-time
verifiers reading one concatenated string: Merlin's proof of $q(n)$ bits and Arthur's
$p(n)$ coins, in that order for MA and in the reverse order for AM, with
$p,q\in\mathbb{N}[X]$ and exact message lengths. Completeness is at least $2/3$ and
soundness at most $1/3$, stated as counts of random strings. Both classes are closed under
polynomial-time many-one reductions. Majority votes reduce the error to $2^{-e(n)}$ for
every polynomial $e$; for MA the proof length is fixed before $e$ is chosen, and for AM
Arthur's rows are answered in parallel. The $\Sigma_2$ and $\Pi_2$ placements use
Lautemann's covering lemma with $12p+1$ shifts after amplification; MA $\subseteq$ SBP
counts accepted proof--coin pairs. Interactive-machine characterizations, round collapse
$\cc{AM}[k]=\cc{AM}$ and $\cc{SBP}\subseteq\cc{AM}$ are not formalized.
\citep{Babai1985,BabaiMoran1988,BoehlerGlasserMeister2006}\citep[Section 8.2]{AroraBarak09}

\begin{result}
\subsection{Toda's theorem}
\[
\cc{PH}\subseteq\cc{P}^{\#\cc{P}}.
\]
Here $\#\cc{P}$ counts witnesses of polynomial-time verifiers. The target class
uses a polynomial-time oracle program answering decision problems via bits of
one such counting function. The program is fixed before the oracle is chosen
and must work for every oracle language agreeing on valid encoded queries;
off-code answers are arbitrary. The target class has no separate query-count
parameter. The nondeterministic-machine characterization of $\#\cc{P}$
remains separate unfinished work.\citep{Toda1991}
\decl{Complexity/Computability/Complexity/Classes/Toda.lean}{121}{Cslib.Complexity.PH_subset_PWithSharpPOracle}
\end{result}

\begin{result}
\subsection{Adleman's theorem}
\[
\cc{BPP}\subseteq\cc{P/poly}.
\]
Every bounded-error polynomial-time language has a nonuniform family of
polynomial-size De Morgan circuits. The result also covers encoded problems by
taking the preimage of a bit-string language. It does not provide a uniform
algorithm to choose the successful random strings or construct the circuits.
\citep{Adleman1978,AroraBarak09}
\decl{Complexity/Computability/Circuit/Boolean/PolynomialTime.lean}{172}{Cslib.Circuits.Boolean.PPoly.exists_of_mem_BPP}
\end{result}

\begin{result}
\subsection{Derandomization from average-case hardness}
Suppose a language $L$ belongs to $\cc{E}=\cc{DTIME}(2^{O(n)})$ and there is a
positive integer $c$ such that, for every sufficiently large $n$, every De Morgan
circuit of size at most $s_n=2^{\lfloor n/c\rfloor}$ agrees with $L$ on at most
$1/2+1/s_n$ of the $n$-bit inputs. Then $\cc{P}=\cc{BPP}$, uniformly for encoded
problems. The Nisan--Wigderson generator and exhaustive seed search have certified
polynomial time. Worst-case exponential circuit hardness alone is not the
hypothesis proved sufficient here.\citep{NisanWigderson1994}
\decl{Complexity/Computability/Complexity/Classes/BPP/Derandomization.lean}{307}{Cslib.Complexity.P_eq_BPP_of_isAverageCaseHard}
\end{result}

\begin{result}
\subsection{Counting algebra and the GapP characterization of PP}
For each fixed encoding, $\#\cc{P}$ is closed under pointwise addition and
multiplication, and $\cc{GapP}$ under addition, subtraction and multiplication;
the constant natural-valued and integer-valued functions belong to the
respective classes. Both classes are also closed under precomposition by a
polynomial-time function between encoded instance types. The resulting count
is exactly the original count on the mapped instance; no padding factor is
introduced. Moreover,
\[
A\in\cc{PP}\quad\Longleftrightarrow\quad
\exists g\in\cc{GapP}\ \forall x,\quad x\in A\ \Longleftrightarrow\ g(x)>0.
\]
Here $\#\cc{P}$ counts accepting certificates of a deterministic polynomial-time
verifier of polynomial certificate length; $\cc{GapP}$ consists of differences
of two such counts. These are verifier characterizations. The corresponding
nondeterministic-machine characterizations remain unfinished.
\citep{Valiant1979,FennerFortnowKurtz1994}
\decl{Complexity/Computability/Complexity/Classes/SharpP/Closure.lean}{105}{Cslib.Complexity.SharpP.subsemiring}
\decl{Complexity/Computability/Complexity/Classes/SharpP/Closure.lean}{201}{Cslib.Complexity.GapP.subring}
\decl{Complexity/Computability/Complexity/Classes/PP/Characterization.lean}{182}{Cslib.Complexity.mem_PP_iff_exists_GapP}
\declat{4f6bdaf35ebe60aec6d1063ae513ac342df13589}{Complexity/Computability/Complexity/Classes/SharpP/Closure.lean}{230}{Cslib.Complexity.SharpP.comp_mem}
\declat{4f6bdaf35ebe60aec6d1063ae513ac342df13589}{Complexity/Computability/Complexity/Classes/SharpP/Closure.lean}{251}{Cslib.Complexity.GapP.comp_mem}
Polynomial-time functions with binary natural output belong to $\#\cc{P}$;
those with binary integer output belong to $\cc{GapP}$. The bound is on output
bit length, so the numeric values may be exponential in input length.
\citep[Theorem 3.5]{Fortnow1997}
\declat{814e76e29fa938afaed4e1bd5763fa490bc39ca9}{Complexity/Computability/Complexity/Classes/SharpP/PolyTime.lean}{41}{Cslib.Complexity.SharpP.of_polyTime}

\end{result}

\begin{result}
\subsection{Boolean closure and unambiguous polynomial time}
For every fixed injective encoding, $\cc{P}$, $\cc{NP}$ and $\cc{coNP}$ are
closed under union and intersection. The odd-acceptance class $\oplus\cc{P}$
is closed under union, intersection and symmetric difference.
The verifier class $\cc{UP}$ satisfies
\[
\cc{P}\subseteq\cc{UP}\subseteq\cc{NP}\cap\oplus\cc{P}.
\]
Its verifier has at most one accepting certificate of the specified polynomial
length on every input, and exactly one on members. Equivalently, a
$\#\cc{P}$ function bounded by one is one precisely on the language.
UP is closed under polynomial-time many-one reductions. This establishes the
verifier formulation; an equivalence with unambiguous nondeterministic machines
is not asserted.\citep{AroraBarak09,Papadimitriou94}
The complement class $\cc{coUP}$ contains P and is contained in both coNP
and $\oplus\cc{P}$; it also inherits polynomial-time reduction closure.
These are elementary consequences of the complement convention and the
corresponding UP containments.

\begin{result}
\subsection{Exact counting and PP}
For the same verifier-based counting functions, the class $\cc{C}_{=}\cc{P}$
is characterized by
\[
A\in\cc{C}_{=}\cc{P}\quad\Longleftrightarrow\quad
\exists g\in\cc{GapP}\ \forall x,\quad x\in A\ \Longleftrightarrow\ g(x)=0.
\]
Equivalently, membership is equality of two $\#\cc{P}$ functions. This class
is closed under union, intersection and polynomial-time many-one reductions,
and satisfies
\[
\cc{coNP}\subseteq\cc{C}_{=}\cc{P}\subseteq\cc{PP}.
\]
The complement class $\cc{coC}_{=}\cc{P}$ consists of nonzero sets of GapP
functions and satisfies $\cc{NP}\subseteq\cc{coC}_{=}\cc{P}\subseteq\cc{PP}$.
It inherits union, intersection and reduction closure. This does not assert
$\cc{C}_{=}\cc{P}=\cc{coC}_{=}\cc{P}$ or a counting-machine characterization.

\begin{result}
\subsection{Bounded ambiguity and small gaps}
For each injective input encoding, FewP consists of languages recognized by
positive $\#\cc{P}$ counts bounded on every input by a polynomial in encoded
input length. The certificate-length polynomial and ambiguity-bound polynomial
are separate. SPP consists of languages whose characteristic function is in
GapP: its gap is exactly one on members and zero on nonmembers. Then
\[
\cc{UP}\subseteq\cc{FewP}\subseteq\cc{NP},\qquad
\cc{UP}\cup\cc{coUP}\subseteq\cc{SPP}
\subseteq\cc{C}_{=}\cc{P}\cap\cc{coC}_{=}\cc{P}.
\]
FewP is closed under union, intersection and polynomial-time many-one reductions;
SPP has these closures and complement closure. The elementary proofs reuse the
count algebra: FewP union adds ambiguity bounds and intersection multiplies them;
SPP complement subtracts its gap from one and intersection multiplies gaps.
These are verifier-based results; neither a machine characterization nor the
stronger FewP--SPP containment is established by this entry.
The FewP definition follows Allender--Rubinstein and the explicit acceptance-set
formulation of Borchert--Hemaspaandra--Rothe; the SPP definition and UP inclusion
follow Fenner--Fortnow--Kurtz and Fortnow's survey.
\citep{AllenderRubinstein1988}\citep[Section 1]{BorchertHemaspaandraRothe2000}
\citep{FennerFortnowKurtz1994}\citep[Classification 4.4 and Corollary 4.11]{Fortnow1997}

\begin{result}
\subsection{Modular counting}
For every fixed natural modulus $k\ge2$, $\cc{Mod}_{k}\cc{P}$ consists of
languages with a $\#\cc{P}$ witness count $f$ such that
\[
x\in A\quad\Longleftrightarrow\quad f(x)\not\equiv0\pmod{k}.
\]
The modulus is a class parameter, not an input. We prove
$\cc{P}\subseteq\cc{UP}\subseteq\cc{Mod}_{k}\cc{P}$,
$\cc{Mod}_{2}\cc{P}=\oplus\cc{P}$, and closure under polynomial-time
many-one reductions. For prime $k$, the class is also closed under complement,
union and intersection. The complement proof applies Fermat's little theorem
to the count $1+(k-1)f(x)^{k-1}$; intersection uses multiplication and primality.
No corresponding composite-modulus Boolean closure is asserted.
The total Lean definition also covers the degenerate moduli: at zero it equals
NP, and at one only the empty language belongs. The standard $k\ge2$ convention
and nonzero-residue acceptance follow Fortnow's Classification 4.7; the closure
proofs are elementary algebraic consequences of the existing counting APIs.
\citep[Classification 4.7]{Fortnow1997}\citep{PapadimitriouZachos1983}

\begin{result}
\subsection{Wide counting gaps}
For every fixed injective encoding,
\[
\cc{SPP}\subseteq\cc{LWPP}\subseteq\cc{WPP}
\subseteq\cc{C}_{=}\cc{P}\cap\cc{coC}_{=}\cc{P}.
\]
In WPP there are a GapP function $g$ and a polynomial-time, binary-integer
function $f$, nonzero on every input, with $g(x)=f(x)$ on members and $g(x)=0$
on nonmembers. LWPP restricts the target to $f(|x|)$, with $f$ computable from
unary length and nonzero on every natural argument, including zero. Negative
and exponentially large targets are allowed. Both classes are closed under
complement, union and intersection; WPP also has polynomial-time many-one
reduction closure. The proofs use exact-gap algebra and the FP-to-GapP bridge.
These verifier-based statements follow the definitions of Fenner--Fortnow--Kurtz,
as restated by Hemaspaandra et al.; they do not establish counting-machine
characterizations, oracle lowness, or robustness under multiple target values.

\begin{result}
\subsection{Small bounded-error and almost-wide counting}
For every fixed injective encoding,
\[
\cc{NP}\cup\cc{BPP}\subseteq\cc{SBP}\subseteq\cc{PP},
\qquad
\cc{SPP}\cup\cc{BPP}\subseteq\cc{AWPP}\subseteq\cc{PP}.
\]
SBP has a verifier-based \#P function $f$ and a polynomial-time function
$k$ with binary natural output such that $f(x)\geq k(x)$ on members and
$2f(x)<k(x)$ on nonmembers. The nonnegative threshold may be zero on
members; requiring it to be positive on every input gives the same class.
AWPP has a verifier-based GapP function $g$ and $p\in\mathbb{N}[X]$ with
$0\leq g(x)\leq 2^{p(|x|)}$ on every input, and
\[
x\in A\Rightarrow 3g(x)\geq 2\cdot 2^{p(|x|)},\qquad
x\notin A\Rightarrow 3g(x)\leq 2^{p(|x|)}.
\]
Both classes are closed under polynomial-time many-one reductions between
injectively encoded types. AWPP is closed under complement; SBP is closed
under union. The AWPP reduction proof permits an input-dependent, unary
polynomial-time exponent and pads it to a polynomial in encoded input length.
These are elementary consequences of the existing counting algebra and BPP
amplification. The definitions follow Watson's formulation of the class
introduced by B\"ohler--Gla{\ss}er--Meister, and Fenner's constant-error
characterization of AWPP.\citep{BoehlerGlasserMeister2006}
\citep[Definition 2]{Watson2012}\citep{Fenner2003}
Counting-machine characterizations, $\cc{SBP}\subseteq\cc{AM}$,
uniform AWPP amplification to input-dependent error and PP-lowness remain
outstanding. The following entry adds fixed amplification and Boolean closure.

\begin{result}
\subsection{Wide gaps, amplification and Boolean closure}
For every fixed injective encoding,
\[
\cc{LWPP}\subseteq\cc{WPP}\subseteq\cc{AWPP}.
\]
The WPP target may be signed, input-dependent and exponentially large: it is a
nonzero, binary-output polynomial-time integer function, with the GapP witness
equal to that target on members and zero on nonmembers.
The classical containment is credited to Fenner--Fortnow--Kurtz--Li;
this formalization squares the target and uses binary length to obtain a dyadic
normalization.\citep[Section 6.1]{FennerEtAl2003}

Every AWPP language has a GapP witness $g$ and $p\in\mathbb{N}[X]$ such that,
writing $D(x)=2^{p(|x|)}$, one has $0\leq g(x)\leq D(x)$ and
\[
x\in A\Rightarrow 7D(x)\leq8g(x),\qquad
x\notin A\Rightarrow 8g(x)\leq D(x).
\]
Three fixed rounds of the majority-of-three polynomial give this error bound;
the denominator exponent is multiplied by $27$. This is a fixed-size instance
of Fenner's amplification method.\citep{Fenner2003}
AWPP is closed under intersection, union, difference and symmetric difference
of two languages over the same encoding, as well as complement.

For SBP, every fixed integer $m>0$ admits a \#P witness $f$ and a positive
binary-output polynomial-time natural threshold $k$ with
\[
x\in A\Rightarrow k(x)\leq f(x),\qquad
x\notin A\Rightarrow 2^m f(x)<k(x).
\]
This uses fixed powers as in Watson's gap strengthening; $m$ is fixed
independently of the input.\citep[Section 1.1]{Watson2012}
Also $A\in\cc{SBP}$ and $B\in\cc{BPP}$ imply $A\cap B\in\cc{SBP}$,
using the existing input-dependent BPP error reduction and a dyadic bound on
\#P counts. This does not establish intersection closure for two SBP languages.
All statements use verifier-based \#P and GapP; their nondeterministic-machine
characterizations and uniform input-dependent AWPP amplification remain open
formalization work.

\subsection{Polynomial-size circuits for the determinant}
For every commutative ring $K$ and $n\in\mathbb N$, the generic $n\times n$
determinant has a division-free arithmetic circuit over $K$ of size at most
\[
3n^4+4.
\]
Consequently its polynomial family belongs to $\cc{VP}$ over $K$. The same circuit
evaluates determinants over every commutative $K$-algebra, with matrix entries
supplied in row-major order. This is a gate-count assertion, not a certified
uniform circuit-generation bound.\citep{Bird2011,Valiant1979Algebra}

\subsection{Polynomial-time division-free integer determinant}
Represent an integer matrix as a list of rows $A\in\mathrm{List}(\mathrm{List}\,\mathbb Z)$ encoded by
$\texttt{encList}\;(\texttt{encList}\;\texttt{encIntBin})$, where $\texttt{encIntBin}$ encodes each
$z\in\mathbb Z$ in sign-magnitude binary using $\texttt{intBits}(z)=\mathrm{size}(|z|)+1$ bits.
Writing $d=|A|$ for the number of rows, missing entries at $(i,j)$ read as $0$ and extra entries outside
$d\times d$ are ignored, denoting $\texttt{ofRows}(A)\in\mathbb Z^{d\times d}$. The staged Bird recurrence
$\texttt{detRows}(A)$ iterates $d-1$ stages $\texttt{stepRows}(A)$ from the squared $d\times d$ table
$\texttt{squareRows}(A)$, returning $1$ when $d=0$ and
$(-1)^{d-1}\texttt{rowEntry}(\texttt{stagesRows}(A,d-1),0,0)$ when $d\ge 1$. On every list of rows $A$,
\[
\texttt{detRows}(A)=\det(\texttt{ofRows}(A)),
\]
and in particular $\texttt{detRows}(\texttt{rowsOf}(M))=\det(M)$ for every $M\in\mathbb Z^{d\times d}$.
If every entry of $A$ satisfies $|A_{i,j}|<2^b$ (equivalently $\texttt{intBits}(A_{i,j})\le b+1$), every
entry of stage $k$ has absolute value strictly below $2^{b+k(1+d+b)}$ and bit length at most
$b+k(1+d+b)+1$. For encoded input length $L=|\texttt{encList}(\texttt{encList}\;\texttt{encIntBin})\,A|$ and every
$i\le k$, the encoded stage table $\texttt{stagesRows}(A,i)$ has length at most
$\texttt{stageTableBound}(L,k)=L(2L(2(L+k(2L+1)+1)+1)+1)$, and $\texttt{detRows}$ is tidily computable in
polynomial time and space on a multi-tape Turing machine:
\[
\texttt{ComputableTidilyInPolyTime}\;\texttt{detRows}\;(\texttt{encList}\;(\texttt{encList}\;\texttt{encIntBin}))\;\texttt{encIntBin}.
\]
This certifies the integer determinant stage of the rational matrix hitting-list runtime; the
full generator runtime is assembled in the rational hitting-list entry below.\citep{Bird2011,OpenAI2026RationalHitting,Schlesinger2026RationalHitting}

\begin{result}
\subsection{Circuit mass production with leading coefficient $1/(1-\gamma)$}
Fix real $0\le\gamma<1$ and $\varepsilon>0$. For every sufficiently large $n$,
every Boolean function $f$ on $n$ bits, and every integer $1\le k\le2^{\gamma n}$,
one circuit computes $f$ independently on $k$ disjoint $n$-bit input blocks with
nonconstant-gate cost at most
\[
\left(\frac{1}{1-\gamma}+\varepsilon\right)\frac{2^n}{n}.
\]
The cutoff depends on $\gamma,\varepsilon$, not on $f,k$. Here NOT, AND and OR
each cost one and constants cost zero, unlike the total-gate convention used in
other entries. The coefficient is optimal at $\gamma=0$, where a single copy of
most functions already needs $(1-o(1))2^n/n$ nonconstant gates\citep{Shannon1949}\citep[Theorem 1.14]{Jukna2012};
no matching lower bound is claimed for $\gamma>0$. The construction is nonuniform and supplies no
certified runtime for choosing its tables.\citep{Schlesinger2026MassProduction}

\end{result}

\subsection{Polynomial-time binary division, Euclidean GCD, and reduced-rational encoding}
Over the unsigned little-endian binary natural encoding \texttt{encNat} and sign-magnitude
binary integer encoding \texttt{encIntBin}, the following operations are computable tidily in
polynomial time on multi-tape Turing machines
(\texttt{Turing.MultiTapeTM.ComputableTidilyInPolyTime}):
\begin{itemize}
\item \textbf{Binary division and remainder.} Folding the shift-and-subtract step
\texttt{BinaryDiv.divModStep} over any most-significant-bit-first word $w\in\{0,1\}^*$ from
$(0,0)$ produces $(\mathrm{val}(w)/b,\mathrm{val}(w)\bmod b)$, with encoded pair length at most
$3|w|+1$. On the reversed digit list of $a\in\mathbb N$ this yields $(a/b,a\bmod b)$ (including
$a/0=0$ and $a\bmod 0=a$), certifying $(a,b)\mapsto a/b$ and $(a,b)\mapsto a\bmod b$.
\item \textbf{Euclidean greatest common divisor.} For $0<a\le b$, the remainder step satisfies
$2a(b\bmod a)\le ab$. Iterating \texttt{BinaryGcd.gcdStep} for
$\ell(a)+\ell(b)+1$ steps from $(a,b)$, where $\ell(n)=|\texttt{Nat.bits}\ n|$, reaches
$(\gcd(a,b),0)$ while keeping every intermediate encoded pair within three times the input pair
length, certifying $(a,b)\mapsto\gcd(a,b)$.
\item \textbf{Canonical natural and reduced-rational encodings.} The self-delimiting natural code
$\texttt{encodeNat}(a)=\texttt{true}^{\ell(a)}\mathbin{+\!+}[\texttt{false}]\mathbin{+\!+}\texttt{Nat.bits}(a)$
is computable in polynomial time from \texttt{encNat}. For $a,b\in\mathbb Z$ with $b\ne 0$ and
$g=\gcd(|a|,|b|)$, the reduced rational $q=\texttt{Rat.divInt}(a,b)$ has numerator magnitude
$|q.\texttt{num}|=|a|/g$, denominator $q.\texttt{den}=|b|/g$, and negativity bit
$(a\ne 0)\wedge((a<0)\oplus(b<0))$ (and $\texttt{Rat.divInt}(a,0)=0$), certifying the canonical
reduced-rational encoding $(a,b)\mapsto\texttt{encodeRat}(\texttt{Rat.divInt}(a,b))$ from
$\texttt{encPair}\ \texttt{encIntBin}\ \texttt{encIntBin}$ to \texttt{encBits}.
\end{itemize}
These certificates cover individual natural and reduced-rational encodings and feed the
polynomial-time rational matrix hitting-list generator below.

\subsection{Polynomial-time rational matrix hitting-list generator}
Let $n,s\in\mathbb N$. A noncommutative rational formula $f\in\texttt{Formula}\ n$ is
\emph{admissible} if it has a defined value at some substitution of $d\times d$ rational matrices
with $d>0$, and \emph{nonzero} if it has a defined nonzero value at some such substitution. The
executable hitting list $\texttt{hittingList}(n,s)\in\texttt{Output}\ n$ consists of substitutions
of $d\times d$ rational matrices with common dimension
\[
0 < d = \texttt{matrixDimension}(e(n,s)) \le 256(n+s+1)^{10}
\]
and canonical reduced-rational binary encoding length
\[
|\texttt{encodeOutput}(\texttt{hittingList}(n,s))| \le
\texttt{PolynomialBounds.outputConstant}\cdot(n+s+1)^{104}
\]
such that every admissible nonzero formula $f\in\texttt{Formula}\ n$ with
$\texttt{nodeCount}(f)\le s$ evaluates to an invertible matrix at some tuple
$X\in(\texttt{hittingList}(n,s)).\texttt{tuples}$ (\texttt{hittingList\_hits}).

The runtime pipeline realizes this construction on lists of rows of binary sign-magnitude integers
(\texttt{encIntBin}) and certifies every stage in
\texttt{Turing.MultiTapeTM.ComputableTidilyInPolyTime}:
\begin{itemize}
\item \textbf{Parameter schedule and integer lift tables.} From $(n,s)$ in unary,
\texttt{generatorParameters} computes the truncation $N$, exponent $e$, ambient dimension $M=2^{e+1}$,
cyclotomic degree $D_g=2^e$, grid size $H$, and matrix dimension $d=MD_g$ in unary. At each grid
point $(t,p,r)$, $\texttt{liftRows}(n,N,M,D_g,t,p,r)$ computes the $n+1$ integer lift tables
$[U_0,U_1,\dots,U_n]$ by division-free shift-and-add binomial recurrences (\texttt{coeffRow}, with
encoded intermediate length bounded by \texttt{coeffRowBound}), integer Taylor test tables
(\texttt{testRows}), and double-sum cyclotomic regular-representation lifts (\texttt{liftTable}),
agreeing with \texttt{ExecutableGenerator.inputTables} (\texttt{liftRows\_eq\_inputTables}). At the
generator's parameters for $t,p,r\le H$, every entry $z$ of every table in \texttt{liftRows}
satisfies $\texttt{intBits}(z)\le\texttt{logConstant}\cdot(n+s+1)^{22}+2$.
\item \textbf{Per-point Cramer normalization and grid enumeration.} Given integer tables
$\texttt{tables}=[U_0,U_1,\dots,U_n]$, \texttt{pointWordOfTables} and \texttt{pointWordsOfTables}
compute $\mathrm{den}=\texttt{detRows}(U_0)$, emit $[]$ when $\mathrm{den}=0$, and otherwise emit
the reduced-rational encoding of every entry $(U_0^{-1}U_a)_{i,j}$ as
$\texttt{encodeRat}(\texttt{Rat.divInt}(\texttt{detRows}(\texttt{cramerRows}(U_0,U_a,i,j)),\mathrm{den}))$,
agreeing with \texttt{ExecutableGenerator.atPoint} composed with \texttt{encodeTuple}
(\texttt{pointWord\_eq}, \texttt{pointWords\_eq}). Enumerating $\{1,\dots,H\}^3$ in lexicographic
order via \texttt{gridPointWords} and prepending $\texttt{encodeNat}(MD_g)$ and the encoded emitted
tuple count produces $\texttt{hittingListWord}(n,s)=\texttt{encodeOutput}(\texttt{hittingList}(n,s))$
(\texttt{hittingListWord\_eq}), certifying $(n,s)\mapsto\texttt{encodeOutput}(\texttt{hittingList}(n,s))$
from $\texttt{encPair}\ \texttt{encUnary}\ \texttt{encUnary}$ to \texttt{encBits}
(\texttt{hittingList\_polytime}).
\item \textbf{Standalone bit-word generator.} Writing
$\texttt{inputWord}(n,s)=\texttt{true}^n\mathbin{+\!+}[\texttt{false}]\mathbin{+\!+}\texttt{true}^s$
of length $n+s+1$, the parser \texttt{parseInput} satisfies
$\texttt{parseInput}(w)=\texttt{some}(n,s)\iff 1\le n\wedge 1\le s\wedge w=\texttt{inputWord}(n,s)$.
The standalone generator
\[
\texttt{hittingGenerator} : \mathrm{List}\,\mathrm{Bool} \to \mathrm{List}\,\mathrm{Bool}
\]
satisfies $\texttt{hittingGenerator}(\texttt{inputWord}(n,s))=\texttt{encodeOutput}(\texttt{hittingList}(n,s))$
for all $1\le n$ and $1\le s$, returns $[]$ on all other words, and is computable tidily in
polynomial time on a multi-tape Turing machine (\texttt{hittingGenerator\_polytime},
\texttt{hittingGenerator\_spec}).
\end{itemize}
This certifies polynomial-time generation of the hitting list; polynomial bit complexity of
evaluating a rational formula on the emitted matrices is not asserted here.

\subsection{Polynomial identity testing, Schwartz--Zippel, and Kabanets--Impagliazzo}
A single-output integer arithmetic circuit bundled with its input arity
$\langle n,c\rangle\in\Sigma n,\;\texttt{Circuit}\;(\texttt{arithSignature}\;\mathbb Z)\;n\;1$ is
encoded injectively as a bit word by $\texttt{encArithCircuitInt}$ (input arity $n$ in unary via
$\texttt{encUnary}$, gate operations with sign-magnitude binary integer constants via $\texttt{encIntBin}$,
and wire indices in binary via $\texttt{encNat}$), with $n\le|\texttt{encArithCircuitInt}\langle n,c\rangle|$
(\texttt{le\_length\_encArithCircuitInt}). The polynomial identity testing language over $\mathbb Z$ is
\[
\texttt{PITInt}
=
\bigl\{\langle n,c\rangle \;\big|\; c.\texttt{polynomial}\;0 = 0 \in \texttt{MvPolynomial}\;(\mathrm{Fin}\;n)\;\mathbb Z\bigr\}.
\]
Three components connect $\texttt{PITInt}$ to randomized zero testing and circuit lower bounds:
\begin{itemize}
\item \textbf{Multivariate Schwartz--Zippel root bound and hitting property.} Over any integral domain $R$,
every nonzero polynomial $P\in\texttt{MvPolynomial}\;(\mathrm{Fin}\;n)\;R$ and finite subset $S\subseteq R$
satisfy
\[
|S|\cdot\bigl|\{x\in S^n \mid P(x)=0\}\bigr| \le \deg(P)\cdot |S|^n
\quad\text{and}\quad
\bigl|\{x\in S^n \mid P(x)=0\}\bigr| \le \deg(P)\cdot |S|^{n-1},
\]
so if $\deg(P)<|S|$ there exists $x\in S^n$ with $P(x)\ne 0$. Specializing to $c.\texttt{polynomial}\;0$
shows that any integer circuit $\langle n,c\rangle\notin\texttt{PITInt}$ vanishes at at most
$\deg(c.\texttt{polynomial}\;0)\cdot |S|^{n-1}$ points of $S^n$ and evaluates under
$\texttt{arithInterpretation}\;\mathbb Z$ to a nonzero integer at some point of $S^n$ whenever
$\deg(c.\texttt{polynomial}\;0)<|S|$.
\item \textbf{Laplace cofactor self-reducibility of the permanent to $\texttt{PITInt}$.} Over any
commutative semiring $R$, the permanent of $A\in R^{(n+1)\times(n+1)}$ expands along column~$0$ or row~$0$:
\[
\mathrm{perm}(A) = \sum_{j=0}^{n} A_{0,j}\,\mathrm{perm}(A_{\mathrm{succ},\,j.\mathrm{succAbove}}).
\]
On the generic matrix $\texttt{genericMatrix}\;R\;(m+1)$, the $(0,j)$-minor permanent equals
$\texttt{rename}\;(\texttt{minorVar}\;j)\;\mathrm{perm}(\texttt{genericMatrix}\;R\;m)$. From candidate
circuits $c_0$ on $0$ variables, $c_m$ on $m^2$ variables, and $c_{m+1}$ on $(m+1)^2$ variables, the
constructive difference circuits $\texttt{permBaseDiff}(c_0)$ (of gate size $|c_0|+4$, computing
$c_0.\texttt{polynomial}\;0 - 1$) and $\texttt{permStepDiff}(m,c_m,c_{m+1})$ (of gate size
$|c_{m+1}|+(m+1)(|c_m|+2)+4$, computing
$c_{m+1}.\texttt{polynomial}\;0 - \sum_{j=0}^{m} X_{(0,j)}\,\texttt{rename}(\texttt{minorVar}\;j)(c_m.\texttt{polynomial}\;0)$)
satisfy the downward self-reducibility equivalence of Kabanets and Impagliazzo\citep[Lemma 4.2]{KabanetsImpagliazzo2004}:
for every family $C : \forall m,\;\texttt{Circuit}\;(\texttt{arithSignature}\;\mathbb Z)\;(m^2)\;1$ and $n\in\mathbb N$,
\[
\bigl(\forall m\le n,\; (C\;m).\texttt{polynomial}\;0 = \mathrm{perm}(\texttt{genericMatrix}\;\mathbb Z\;m)\bigr)
\iff
\langle 0,\texttt{permBaseDiff}(C\;0)\rangle\in\texttt{PITInt}
\;\wedge\;
\forall m<n,\;\langle (m+1)^2,\texttt{permStepDiff}(m,C\;m,C(m+1))\rangle\in\texttt{PITInt}.
\]
The explicit recursive family $\texttt{laplacePermCircuit}$ unconditionally satisfies these identities at
every dimension $n$ (\texttt{isLaplacePermChain\_laplacePermCircuit}), and if
$\texttt{PITInt}\in\cc{P}\;\texttt{encArithCircuitInt}$ and the integer permanent p-family
$\texttt{intPermFamily}=\texttt{permFamily}\;\mathbb Z$ belongs to $\cc{VP}\;\mathbb Z$, there exist a
polynomial-time decider for $\texttt{PITInt}$ and a polynomially bounded permanent circuit family with
polynomially bounded step-test sizes certified by it (\texttt{PermPITCertified},
\texttt{exists\_certified\_perm\_family\_of\_PITInt\_mem\_P\_of\_mem\_VP}). The gate-size bounds on
$\texttt{permBaseDiff}$ and $\texttt{permStepDiff}$ are combinatorial circuit sizes; uniform Turing-machine
construction of $\texttt{permStepDiff}$ from encoded circuits is not separately certified here.
\item \textbf{Unconditional $\cc{NEXP}$ lower bound and conditional Kabanets--Impagliazzo consequence.}
Combining $\cc{\Sigma}_k^{\cc{P}}\subseteq\cc{PSPACE}\subseteq\cc{EXP}\subseteq\cc{NEXP}$ with Kannan's
theorem shows unconditionally that for every $k\in\mathbb N$ there is a language $L\in\cc{NEXP}$ outside
$\cc{SIZE}(c(n+1)^k)$ for all $c\in\mathbb N$ (\texttt{exists\_mem\_NEXP\_notMem\_SIZE}). If
$\texttt{PITInt}\in\cc{P}\;\texttt{encArithCircuitInt}$, then under the explicit collapse
hypothesis from $\cc{NEXP}\subseteq\cc{P/poly}$ and $\texttt{PermPITCertified}$ (either to a fixed degree
$k$ such that every $L\in\cc{\Sigma}_2^{\cc{P}}$ lies in $\bigcup_c\cc{SIZE}(c(n+1)^k)$, refuted by Kannan's
theorem, or to $\cc{EXP}\subseteq\cc{P}$, refuted by $\cc{P}\subsetneq\cc{EXP}$), either
$\cc{NEXP}\not\subseteq\cc{P/poly}$ or $\texttt{intPermFamily}\notin\cc{VP}\;\mathbb Z$. The easy-witness
and Valiant--Toda collapse step from $\cc{NEXP}\subseteq\cc{P/poly}$ and $\texttt{PermPITCertified}$ is
taken as an explicit hypothesis.\citep{DeMilloLipton1978,Zippel1979,Schwartz1980,Kannan1982,KabanetsImpagliazzo2004,AroraBarak09}
\end{itemize}

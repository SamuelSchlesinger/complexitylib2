\section{Arithmetic circuits and simultaneous computation}

\begin{result}
\subsection{Polynomial-size circuits for the determinant}
For every commutative ring $K$ and $n\in\mathbb N$, the generic $n\times n$
determinant has a division-free arithmetic circuit over $K$ of size at most
\[
3n^4+4.
\]
Consequently its polynomial family belongs to $\cc{VP}$ over $K$. The same circuit
evaluates determinants over every commutative $K$-algebra, with matrix entries
supplied in row-major order. This is a gate-count assertion, not a certified
uniform circuit-generation bound.\citep{Bird2011,Valiant1979Algebra}

\begin{result}
\subsection{Polynomial-time division-free integer determinant}
Represent an integer matrix as a list of rows $A\in\mathrm{List}(\mathrm{List}\,\mathbb Z)$ encoded by
$\texttt{encList}\;(\texttt{encList}\;\texttt{encIntBin})$, where $\texttt{encIntBin}$ encodes each
$z\in\mathbb Z$ in sign-magnitude binary using $\texttt{intBits}(z)=\mathrm{size}(|z|)+1$ bits.
Writing $d=|A|$ for the number of rows, missing entries at $(i,j)$ read as $0$ and extra entries outside
$d\times d$ are ignored, denoting $\texttt{ofRows}(A)\in\mathbb Z^{d\times d}$. The staged Bird recurrence
$\texttt{detRows}(A)$ iterates $d-1$ stages $\texttt{stepRows}(A)$ from the squared $d\times d$ table
$\texttt{squareRows}(A)$, returning $1$ when $d=0$ and
$(-1)^{d-1}\texttt{rowEntry}(\texttt{stagesRows}(A,d-1),0,0)$ when $d\ge 1$. On every list of rows $A$,
\[
\texttt{detRows}(A)=\det(\texttt{ofRows}(A)),
\]
and in particular $\texttt{detRows}(\texttt{rowsOf}(M))=\det(M)$ for every $M\in\mathbb Z^{d\times d}$.
If every entry of $A$ satisfies $|A_{i,j}|<2^b$ (equivalently $\texttt{intBits}(A_{i,j})\le b+1$), every
entry of stage $k$ has absolute value strictly below $2^{b+k(1+d+b)}$ and bit length at most
$b+k(1+d+b)+1$. For encoded input length $L=|\texttt{encList}(\texttt{encList}\;\texttt{encIntBin})\,A|$ and every
$i\le k$, the encoded stage table $\texttt{stagesRows}(A,i)$ has length at most
$\texttt{stageTableBound}(L,k)=L(2L(2(L+k(2L+1)+1)+1)+1)$, and $\texttt{detRows}$ is tidily computable in
polynomial time and space on a multi-tape Turing machine:
\[
\texttt{ComputableTidilyInPolyTime}\;\texttt{detRows}\;(\texttt{encList}\;(\texttt{encList}\;\texttt{encIntBin}))\;\texttt{encIntBin}.
\]
This certifies the integer determinant stage of the rational matrix hitting-list runtime; the remaining
generator runtime milestones (certified integer lift tables, per-point
Cramer words, and grid enumeration) remain open.\citep{Bird2011,OpenAI2026RationalHitting,Schlesinger2026RationalHitting}

\begin{result}
\subsection{Circuit mass production with leading coefficient $1/(1-\gamma)$}
Fix real $0\le\gamma<1$ and $\varepsilon>0$. For every sufficiently large $n$,
every Boolean function $f$ on $n$ bits, and every integer $1\le k\le2^{\gamma n}$,
one circuit computes $f$ independently on $k$ disjoint $n$-bit input blocks with
nonconstant-gate cost at most
\[
\left(\frac{1}{1-\gamma}+\varepsilon\right)\frac{2^n}{n}.
\]
The cutoff depends on $\gamma,\varepsilon$, not on $f,k$. Here NOT, AND and OR
each cost one and constants cost zero, unlike the total-gate convention used in
other entries. The coefficient is optimal at $\gamma=0$, where a single copy of
most functions already needs $(1-o(1))2^n/n$ nonconstant gates\citep{Shannon1949}\citep[Theorem 1.14]{Jukna2012};
no matching lower bound is claimed for $\gamma>0$. The construction is nonuniform and supplies no
certified runtime for choosing its tables.\citep{Schlesinger2026MassProduction}

\end{result}

\begin{result}
\subsection{Polynomial-time binary division, Euclidean GCD, and reduced-rational encoding}
Over the unsigned little-endian binary natural encoding \texttt{encNat} and sign-magnitude
binary integer encoding \texttt{encIntBin}, the following operations are computable tidily in
polynomial time on multi-tape Turing machines
(\texttt{Turing.MultiTapeTM.ComputableTidilyInPolyTime}):
\begin{itemize}
\item \textbf{Binary division and remainder.} Folding the shift-and-subtract step
\texttt{BinaryDiv.divModStep} over any most-significant-bit-first word $w\in\{0,1\}^*$ from
$(0,0)$ produces $(\mathrm{val}(w)/b,\mathrm{val}(w)\bmod b)$, with encoded pair length at most
$3|w|+1$. On the reversed digit list of $a\in\mathbb N$ this yields $(a/b,a\bmod b)$ (including
$a/0=0$ and $a\bmod 0=a$), certifying $(a,b)\mapsto a/b$ and $(a,b)\mapsto a\bmod b$.
\item \textbf{Euclidean greatest common divisor.} For $0<a\le b$, the remainder step satisfies
$2a(b\bmod a)\le ab$. Iterating \texttt{BinaryGcd.gcdStep} for
$\ell(a)+\ell(b)+1$ steps from $(a,b)$, where $\ell(n)=|\texttt{Nat.bits}\ n|$, reaches
$(\gcd(a,b),0)$ while keeping every intermediate encoded pair within three times the input pair
length, certifying $(a,b)\mapsto\gcd(a,b)$.
\item \textbf{Canonical natural and reduced-rational encodings.} The self-delimiting natural code
$\texttt{encodeNat}(a)=\texttt{true}^{\ell(a)}\mathbin{+\!+}[\texttt{false}]\mathbin{+\!+}\texttt{Nat.bits}(a)$
is computable in polynomial time from \texttt{encNat}. For $a,b\in\mathbb Z$ with $b\ne 0$ and
$g=\gcd(|a|,|b|)$, the reduced rational $q=\texttt{Rat.divInt}(a,b)$ has numerator magnitude
$|q.\texttt{num}|=|a|/g$, denominator $q.\texttt{den}=|b|/g$, and negativity bit
$(a\ne 0)\wedge((a<0)\oplus(b<0))$ (and $\texttt{Rat.divInt}(a,0)=0$), certifying the canonical
reduced-rational encoding $(a,b)\mapsto\texttt{encodeRat}(\texttt{Rat.divInt}(a,b))$ from
$\texttt{encPair}\ \texttt{encIntBin}\ \texttt{encIntBin}$ to \texttt{encBits}.
\end{itemize}
These certificates cover individual natural and reduced-rational encodings; polynomial-time
generation of the full rational matrix hitting list remains separate.
